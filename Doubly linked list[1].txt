#include <iostream>
#include <cstdlib> 
using namespace std;

struct Node {
    int data;
    Node* prev;
    Node* next;
};
void insertAtEnd(Node*& head, int value) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == nullptr) {
        cout << "Memory allocation failed!" << endl;
        return;
    }
    newNode->data = value;
    newNode->prev = nullptr;
    newNode->next = nullptr;
    if (head == nullptr) {
        head = newNode;
        cout << "Inserted " << value << " at the end (it's the first node).\n";
        return;
    }
    Node* temp = head;
    while (temp->next != nullptr) {
        temp = temp->next;
    }
    temp->next = newNode;
    newNode->prev = temp;


    cout << "Inserted " << value << " at the end.\n";
}
void deleteFromBeginning(Node*& head) {
    if (head == nullptr) {
        cout << "List is empty. Cannot delete.\n";
        return;
    }


    Node* temp = head;
    head = head->next;
    if (head != nullptr) {
        head->prev = nullptr;
    }


    free(temp);
    cout << "Node deleted from the beginning.\n";
}
void displayList(Node* head) {
    if (head == nullptr) {
        cout << "List is empty.\n";
        return;
    }
    Node* temp = head;
    cout << "List: ";
    while (temp != nullptr) {
        cout << temp->data << " ";
        temp = temp->next;
    }
    cout << endl;
}
int main() {
    Node* head = nullptr;
    displayList(head);
    insertAtEnd(head, 10);
    insertAtEnd(head, 20);
    insertAtEnd(head, 30);
    displayList(head);
    deleteFromBeginning(head);
    displayList(head);


    deleteFromBeginning(head);
    displayList(head);
    deleteFromBeginning(head);


    return 0;
}
